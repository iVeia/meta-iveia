#
# Top level uEnv.txt shim that loads the images by PN.
#
# Works by setting the ${altpath}, which is used by uEnv.txt as the directory
# of the boot images.  The ${altpath} is set to the PN (5-digit part number) of
# the SoM.
#
# Note: iv_mb_ord is the 5-digit ordinal from the part number.
#
bootenv_cmd=setenv altpath ${iv_mb_ord}; boot
